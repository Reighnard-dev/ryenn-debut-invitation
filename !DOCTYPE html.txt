        function createPetals() {
            const container = document.getElementById('petals-container');
            for (let i = 0; i < 20; i++) {
                const petal = document.createElement('div');
                petal.className = 'petal';
                petal.style.left = Math.random() * 100 + 'vw';
                petal.style.width = (Math.random() * 12 + 8) + 'px';
                petal.style.height = (Math.random() * 16 + 10) + 'px';
                petal.style.animationDelay = (Math.random() * 10) + 's';
                petal.style.animationDuration = (Math.random() * 8 + 8) + 's';
                container.appendChild(petal);
            }
        }

        function updateCountdown() {
            const eventDate = new Date('November 21, 2026 18:00:00').getTime();
            const now = new Date().getTime();
            const diff = eventDate - now;

            if (diff > 0) {
                const days = Math.floor(diff / (1000 * 60 * 60 * 24));
                const hours = Math.floor((diff % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60));
                const minutes = Math.floor((diff % (1000 * 60 * 60)) / (1000 * 60));
                const seconds = Math.floor((diff % (1000 * 60)) / 1000);

                document.getElementById('days').innerText = String(days).padStart(2, '0');
                document.getElementById('hours').innerText = String(hours).padStart(2, '0');
                document.getElementById('minutes').innerText = String(minutes).padStart(2, '0');
                document.getElementById('seconds').innerText = String(seconds).padStart(2, '0');
            }
        }

        function switchTab(key) {
            document.querySelectorAll('.tab-btn').forEach(btn => {
                btn.className = 'tab-btn px-5 py-2.5 rounded-full font-cinzel text-xs font-bold uppercase tracking-wider text-gray-400 hover:text-gold transition duration-300';
            });
            const activeBtn = document.getElementById(`tab-${key}`);
            if (activeBtn) {
                activeBtn.className = 'tab-btn px-5 py-2.5 rounded-full font-cinzel text-xs font-bold uppercase tracking-wider transition duration-300 bg-gold-gradient text-velvet shadow-md';
            }

            const data = programData[key];
            const container = document.getElementById('tab-content');
            
            let gridHtml = data.items.map((item, idx) => `
                <div class="border border-gold/20 p-3 rounded-lg bg-velvet/40 flex items-center space-x-3">
                    <span class="w-6 h-6 rounded-full bg-gold/10 text-gold font-cinzel text-xs flex items-center justify-center font-bold">${idx + 1}</span>
                    <span class="text-xs text-gray-300 tracking-wide">${item}</span>
                </div>
            `).join('');

            container.innerHTML = `
                <div class="mb-6 text-center">
                    <h3 class="font-cinzel text-xl font-bold text-gold">${data.title}</h3>
                    <p class="text-xs text-gray-400 mt-1">${data.description}</p>
                </div>
                <div class="grid grid-cols-1 sm:grid-cols-2 md:grid-cols-3 gap-3">
                    ${gridHtml}
                </div>
            `;
        }

        function generateQRCode(text) {
            const container = document.getElementById("qrcode");

                colorDark : "#800020",

            });
        }

        function handleRSVP(event) {
            event.preventDefault();
            const name = document.getElementById('guestName').value;
            const message = document.getElementById('guestMessage').value;

            // Update Pass Name
            document.getElementById('passGuestName').innerText = name;
            generateQRCode(`RYENN18-GUEST-${name.replace(/\s+/g, '').toUpperCase()}`);

            // Add wish to Guestbook if provided
            if (message.trim()) {
                const wishesContainer = document.getElementById('wishesContainer');
                const wishCard = document.createElement('div');
                wishCard.className = 'glass-panel p-6 rounded-xl border border-gold/30 animate-fade-in';

                `;
                wishesContainer.prepend(wishCard);

            document.getElementById('pass').scrollIntoView({ behavior: 'smooth' });
        }

        let isPlaying = false;
        function toggleAudio() {
            const audio = document.getElementById('bgAudio');
            const icon = document.getElementById('musicIcon');
            if (isPlaying) {
                audio.pause();
                icon.className = 'fas fa-music text-lg group-hover:rotate-12 transition';
            } else {
                audio.play().catch(e => console.log('Audio playback initialized'));
                icon.className = 'fas fa-pause text-lg text-gold animate-pulse';
            }
            isPlaying = !isPlaying;
        }
